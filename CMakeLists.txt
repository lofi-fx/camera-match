cmake_minimum_required(VERSION 3.20)
project(LoFiFxCameraMatch VERSION 0.10.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(POLICY CMP0169)
  cmake_policy(SET CMP0169 OLD)
endif()
include(FetchContent)
FetchContent_Declare(openfx GIT_REPOSITORY https://github.com/AcademySoftwareFoundation/openfx.git GIT_TAG ab779510b2655b4d11a7e01e5c521f9aa8c88976 GIT_SHALLOW FALSE)
FetchContent_GetProperties(openfx)
if(NOT openfx_POPULATED)
  FetchContent_Populate(openfx)
endif()
add_library(camera_core STATIC src/core/Chart.cpp src/core/Color.cpp src/core/Match.cpp src/core/State.cpp src/core/PatchSampler.cpp src/core/HeroTransfer.cpp)
target_include_directories(camera_core PUBLIC src/core)
add_library(LoFiFxCameraMatch MODULE src/plugin/Plugin.cpp)
target_include_directories(LoFiFxCameraMatch PRIVATE ${openfx_SOURCE_DIR}/include)
target_link_libraries(LoFiFxCameraMatch PRIVATE camera_core)
if(APPLE)
  enable_language(OBJCXX)
  target_sources(LoFiFxCameraMatch PRIVATE src/plugin/MetalRender.mm)
  target_link_libraries(LoFiFxCameraMatch PRIVATE "-framework Metal" "-framework Foundation" "-framework OpenGL")
  set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "" FORCE)
endif()
set_target_properties(LoFiFxCameraMatch PROPERTIES PREFIX "" SUFFIX ".ofx" CXX_VISIBILITY_PRESET default)
set(BUNDLE ${CMAKE_BINARY_DIR}/LoFiFxCameraMatch.ofx.bundle)
add_custom_command(TARGET LoFiFxCameraMatch POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory ${BUNDLE}/Contents/MacOS
  COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:LoFiFxCameraMatch> ${BUNDLE}/Contents/MacOS/LoFiFxCameraMatch.ofx
  COMMAND ${CMAKE_COMMAND} -E copy ${CMAKE_CURRENT_SOURCE_DIR}/Info.plist ${BUNDLE}/Contents/Info.plist)
enable_testing()
add_executable(camera_core_tests tests/CoreTests.cpp)
target_link_libraries(camera_core_tests PRIVATE camera_core)
add_test(NAME camera_core COMMAND camera_core_tests)
if(APPLE)
  add_executable(camera_metal_tests tests/MetalTests.mm src/plugin/MetalRender.mm)
  target_include_directories(camera_metal_tests PRIVATE src/plugin)
  target_link_libraries(camera_metal_tests PRIVATE camera_core "-framework Metal" "-framework Foundation")
  add_test(NAME camera_metal COMMAND camera_metal_tests)
endif()
